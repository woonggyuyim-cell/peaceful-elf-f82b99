const markerColors = { home: "#2878D0", school: "#E69A22", market: "#32986C", hospital: "#D95463", restaurant: "#9364C7" }; function syncMarkers() { if (!map) return; markers.forEach(marker => marker.setMap(null)); markers.clear(); const searchText = $('search').value .trim() .toLowerCase(); for (const place of places) { const position = saved[place.id]; if (!position) continue; const categoryMatches = filter === 'all' || place.type === filter; const searchMatches = (place.name + ' ' + place.area) .toLowerCase() .includes(searchText); if (!categoryMatches || !searchMatches) { continue; } const color = markerColors[place.type] || "#64748B"; const marker = new google.maps.Marker({ position, map, title: place.name, label: { text: place.id, color: "#FFFFFF", fontSize: "11px", fontWeight: "bold" }, icon: { path: google.maps.SymbolPath.CIRCLE, fillColor: color, fillOpacity: 1, strokeColor: "#FFFFFF", strokeWeight: 2, scale: 19 } }); marker.addListener('click', () => { selectPlace(place.id); }); markers.set(place.id, marker); } }
